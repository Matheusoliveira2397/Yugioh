  <button disabled className="w-full py-3 rounded-lg font-bold bg-gray-700 text-gray-500 cursor-not-allowed opacity-60">

  </button>
) : item.type === 'card' ? (

      <button

        disabled={loading || (user.credits || 0) < item.priceCredits}
        className={w-full py-3 rounded-lg font-bold transition-all duration-300 flex items-center justify-center gap-2 text-sm }
      >
        {(user.credits || 0) >= item.priceCredits ? <>💎 Comprar com Créditos</> : <>💎 Insuficiente</>}
      </button>
    )}

        onClick={() => handlePurchase(item, 'cash')}
